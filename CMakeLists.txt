cmake_minimum_required(VERSION 3.21)
project(TechStore VERSION 1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Qt6 6.5 REQUIRED COMPONENTS Quick Sql QuickControls2)
qt_standard_project_setup(REQUIRES 6.5)
qt_add_executable(TechStore main.cpp database/DatabaseManager.h database/DatabaseManager.cpp)
qt_add_qml_module(TechStore URI TechStore VERSION 1.0 QML_FILES qml/Main.qml)
target_link_libraries(TechStore PRIVATE Qt6::Quick Qt6::Sql Qt6::QuickControls2)
set_target_properties(TechStore PROPERTIES WIN32_EXECUTABLE TRUE)
